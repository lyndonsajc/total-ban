  <script>
    const graph = document.getElementById('graph');
    const externalCost = document.getElementById('externalCost');
    const regulatedQ = document.getElementById('regulatedQ');
    const showAreas = document.getElementById('showAreas');

    const externalCostValue = document.getElementById('externalCostValue');
    const regulatedQValue = document.getElementById('regulatedQValue');
    const showAreasValue = document.getElementById('showAreasValue');

    const qRegMetric = document.getElementById('qRegMetric');
    const qOptMetric = document.getElementById('qOptMetric');
    const dwlMetric = document.getElementById('dwlMetric');
    const dwlStatus = document.getElementById('dwlStatus');
    const feedback = document.getElementById('feedback');

    const banBtn = document.getElementById('banBtn');
    const optimalBtn = document.getElementById('optimalBtn');
    const marketBtn = document.getElementById('marketBtn');
    const resetBtn = document.getElementById('resetBtn');

    const W = 820;
    const H = 560;
    const margin = { left: 76, right: 36, top: 35, bottom: 72 };
    const plotW = W - margin.left - margin.right;
    const plotH = H - margin.top - margin.bottom;

    // Linear model designed for clear A-Level teaching diagrams.
    // Quantity range: 0 to 100. Price / benefit-cost range: 0 to 120.
    const qMax = 100;
    const pMax = 120;
    const mpbIntercept = 100;
    const mpbSlope = 1;
    const mpcIntercept = 0;
    const mpcSlope = 1;

    const x = q => margin.left + (q / qMax) * plotW;
    const y = p => margin.top + plotH - (p / pMax) * plotH;

    function MPB(q) { return mpbIntercept - mpbSlope * q; }
    function MSC(q, mec) { return mpcIntercept + mec + mpcSlope * q; }
    function MPC(q) { return mpcIntercept + mpcSlope * q; }

    function clamp(v, lo, hi) { return Math.max(lo, Math.min(hi, v)); }
    function fmt(n) { return Number(n).toFixed(1).replace(/\.0$/, ''); }

    function linePath(fn, mec = 0) {
      const points = [];
      for (let q = 0; q <= 100; q += 1) {
        points.push(`${x(q)},${y(fn(q, mec))}`);
      }
      return `M${points.join(' L')}`;
    }

    function areaBetweenPath(qA, qB, upperFn, lowerFn, mec) {
      const a = clamp(Math.min(qA, qB), 0, 100);
      const b = clamp(Math.max(qA, qB), 0, 100);
      if (Math.abs(a - b) < 0.001) return '';

      const top = [];
      const bottom = [];
      const steps = 28;
      for (let i = 0; i <= steps; i++) {
        const q = a + (b - a) * (i / steps);
        top.push(`${x(q)},${y(upperFn(q, mec))}`);
      }
      for (let i = steps; i >= 0; i--) {
        const q = a + (b - a) * (i / steps);
        bottom.push(`${x(q)},${y(lowerFn(q, mec))}`);
      }
      return `M${top.join(' L')} L${bottom.join(' L')} Z`;
    }

    function welfareAreaToQ(q, mec) {
      // Integral from 0 to q of MPB - MSC.
      // MPB - MSC = (100 - mec) - 2q.
      return (100 - mec) * q - q * q;
    }

    function compute() {
      const mec = Number(externalCost.value);
      const qReg = Number(regulatedQ.value);
      const qMarket = 50; // MPB = MPC: 100 - q = q
      const qOpt = clamp((100 - mec) / 2, 0, 100); // MPB = MSC: 100 - q = q + mec

      const freeMarketDWL = Math.max(0, qMarket - qOpt) ** 2;
      const currentDWL = Math.abs(welfareAreaToQ(qOpt, mec) - welfareAreaToQ(qReg, mec));
      const welfareChange = freeMarketDWL - currentDWL;

      return { mec, qReg, qMarket, qOpt, freeMarketDWL, currentDWL, welfareChange };
    }

    function draw() {
      const { mec, qReg, qMarket, qOpt, freeMarketDWL, currentDWL, welfareChange } = compute();
      externalCostValue.textContent = mec;
      regulatedQValue.textContent = qReg;
      showAreasValue.textContent = showAreas.value === '1' ? 'ON' : 'OFF';
      qRegMetric.textContent = fmt(qReg);
      qOptMetric.textContent = fmt(qOpt);
      dwlMetric.textContent = fmt(currentDWL);

      let status = '';
      if (Math.abs(qReg - qOpt) < 0.5) status = 'No DWL: socially optimal quantity';
      else if (qReg < qOpt) status = 'Welfare loss: regulation too strict';
      else status = 'Welfare loss: overconsumption remains';
      dwlStatus.textContent = status;

      const pMarket = MPC(qMarket);
      const pOpt = MSC(qOpt, mec);
      const pRegMPC = MPC(qReg);
      const pRegMSC = MSC(qReg, mec);

      const gridLines = [0, 20, 40, 60, 80, 100, 120].map(p => `
        <line class="gridline" x1="${margin.left}" y1="${y(p)}" x2="${W - margin.right}" y2="${y(p)}" />
        <text class="small-label" x="${margin.left - 14}" y="${y(p) + 4}" text-anchor="end">${p}</text>
      `).join('');

      const qTicks = [0, 20, 40, 60, 80, 100].map(q => `
        <line class="gridline" x1="${x(q)}" y1="${margin.top}" x2="${x(q)}" y2="${margin.top + plotH}" />
        <text class="small-label" x="${x(q)}" y="${H - 38}" text-anchor="middle">${q}</text>
      `).join('');

      const show = showAreas.value === '1';
      let areas = '';
      if (show) {
        // RED AREA = welfare loss caused by the regulated quantity.
        // If Qreg < Q*, the loss is from useful units not consumed: MPB > MSC.
        // If Qreg > Q*, the loss is from excessive units consumed: MSC > MPB.
        if (qReg < qOpt) {
          areas += `<path class="area-loss" d="${areaBetweenPath(qReg, qOpt, MPB, MSC, mec)}"></path>`;
        } else if (qReg > qOpt) {
          areas += `<path class="area-dwl" d="${areaBetweenPath(qOpt, qReg, MSC, MPB, mec)}"></path>`;
        }
      }

      graph.innerHTML = `
        ${gridLines}
        ${qTicks}
        <line class="axis" x1="${margin.left}" y1="${margin.top + plotH}" x2="${W - margin.right}" y2="${margin.top + plotH}" />
        <line class="axis" x1="${margin.left}" y1="${margin.top}" x2="${margin.left}" y2="${margin.top + plotH}" />

        ${areas}

        <path class="mpb" d="${linePath((q) => MPB(q))}" />
        <path class="msb" d="${linePath((q, m) => MSC(q, m), mec)}" />
        <path class="mpc" d="${linePath((q) => MPC(q))}" />

        <line class="guide" x1="${x(qMarket)}" y1="${y(0)}" x2="${x(qMarket)}" y2="${y(pMarket)}" />
        <line class="guide" x1="${margin.left}" y1="${y(pMarket)}" x2="${x(qMarket)}" y2="${y(pMarket)}" />

        <line class="guide" x1="${x(qOpt)}" y1="${y(0)}" x2="${x(qOpt)}" y2="${y(pOpt)}" />
        <line class="guide" x1="${margin.left}" y1="${y(pOpt)}" x2="${x(qOpt)}" y2="${y(pOpt)}" />

        <line class="regline" x1="${x(qReg)}" y1="${margin.top}" x2="${x(qReg)}" y2="${margin.top + plotH}" />

        <circle cx="${x(qMarket)}" cy="${y(pMarket)}" r="6" fill="#2563eb" />
        <circle cx="${x(qOpt)}" cy="${y(pOpt)}" r="6" fill="#16a34a" />
        <circle cx="${x(qReg)}" cy="${y(pRegMPC)}" r="5" fill="#7c3aed" />
        <circle cx="${x(qReg)}" cy="${y(pRegMSC)}" r="5" fill="#7c3aed" />

        <text class="label" x="${W - 135}" y="${y(MPB(92)) - 8}">MPB</text>
        <text class="label" x="${W - 135}" y="${y(MSC(92, mec)) + 18}">MSC</text>
        <text class="label" x="${W - 140}" y="${y(MPC(92)) - 8}">MPC</text>

        <text class="small-label" x="${x(qMarket)}" y="${H - 18}" text-anchor="middle">Qm ${fmt(qMarket)}</text>
        <text class="small-label" x="${x(qOpt)}" y="${H - 55}" text-anchor="middle">Q* ${fmt(qOpt)}</text>
        <text class="small-label" x="${x(qReg)}" y="${margin.top + 18}" text-anchor="middle" fill="#7c3aed">Q regulated ${fmt(qReg)}</text>

        <text class="label" x="${W / 2}" y="${H - 8}" text-anchor="middle">Quantity of units consumed</text>
        <text class="label" x="22" y="${H / 2}" transform="rotate(-90 22 ${H / 2})" text-anchor="middle">Costs / Benefits ($)</text>
      `;

      const banText = qReg === 0
        ? 'This is a total ban. The red area shows welfare loss from banning units that would have generated net benefit before Q*.'
        : `This is not a total ban. The regulation allows ${fmt(qReg)} units to be consumed.`;

      const compareText = Math.abs(qReg - qOpt) < 0.5
        ? 'The regulated quantity is equal to the socially optimal quantity, so welfare loss is minimised because benefit equals social cost at the margin.'
        : qReg < qOpt
          ? 'The regulated quantity is below the social optimum. The red shaded area shows welfare loss from useful units not consumed, where MPB exceeds MSC.'
          : 'The regulated quantity is above the social optimum. The red shaded area shows welfare loss from excessive units consumed, where MSC exceeds MPB.';

      const welfareText = 'The red shaded area measures the welfare loss from the chosen regulated quantity compared with the social optimum.';

      feedback.innerHTML = `
        <strong>Feedback:</strong> ${banText} ${compareText} ${welfareText}
        <br><br>
        <strong>Original free-market DWL:</strong> ${fmt(freeMarketDWL)}. <strong>Welfare loss from regulated quantity:</strong> ${fmt(currentDWL)}.
      `;
    }

    externalCost.addEventListener('input', draw);
    regulatedQ.addEventListener('input', draw);
    showAreas.addEventListener('input', draw);

    banBtn.addEventListener('click', () => {
